# 1. 安装Medusa CLI

npm install @medusajs/medusa-cli -g


# 2. 创建项目

medusa new my-medusa-store --seed

cd my-medusa-store

# 3. 启动后端服务

npm run start

# 4. 安装前端（可选，用官方React模板）
git clone https://github.com/medusajs/nextjs-starter-medusa.git

cd nextjs-starter-medusa

npm install

npm run dev

# 5. 访问地址

# 后端API：http://localhost:9000

# 前端商城：http://localhost:3000

# 后台管理：http://localhost:7000（需单独安装medusa-admin）
